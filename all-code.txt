'use client'

import { useState, useEffect } from 'react'
import { supabase } from '@/lib/supabase'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { calculateLevel } from '@/lib/levels'

type Tab = 'overview' | 'users' | 'games' | 'recruits' | 'logs'

export default function AdminPage() {
  const router = useRouter()
  const [userProfile, setUserProfile] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  const [isAdmin, setIsAdmin] = useState(false)
  const [busy, setBusy] = useState(false)
  const [msg, setMsg] = useState('')

  const [activeTab, setActiveTab] = useState<Tab>('overview')

  const [stats, setStats] = useState({ users: 0, games: 0, recruits: 0, moments: 0 })
  const [usersList, setUsersList] = useState<any[]>([])
  const [gamesList, setGamesList] = useState<any[]>([])
  const [recruitsList, setRecruitsList] = useState<any[]>([])
  const [logsList, setLogsList] = useState<any[]>([])

  const [userSearch, setUserSearch] = useState('')
  const [gameSearch, setGameSearch] = useState('')
  const [recruitSearch, setRecruitSearch] = useState('')

  useEffect(() => {
    async function checkAuth() {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) {
        alert('鏈櫥褰曪紝鏃犳硶璁块棶鍚庡彴锛?)
        router.push('/login')
        return
      }

      const { data: profile } = await supabase.from('users_profile').select('*').eq('auth_id', user.id).single()

      if (!profile || !['admin', 'super_admin'].includes(profile.role)) {
        alert('鉀?鏉冮檺涓嶈冻锛氫綘鐨勮处鍙锋病鏈夊悗鍙版潈闄?)
        router.push('/')
        return
      }

      setUserProfile(profile)
      setIsAdmin(true)
      await fetchData()
      setLoading(false)
    }
    checkAuth()
  }, [router])

  async function fetchData() {
    const { count: userCount } = await supabase.from('users_profile').select('*', { count: 'exact', head: true })
    const { count: gameCount } = await supabase.from('games').select('*', { count: 'exact', head: true })
    const { count: recruitCount } = await supabase.from('recruits').select('*', { count: 'exact', head: true })
    const { count: momentCount } = await supabase.from('moments').select('*', { count: 'exact', head: true })
    setStats({
      users: userCount || 0,
      games: gameCount || 0,
      recruits: recruitCount || 0,
      moments: momentCount || 0,
    })

    const { data: uData } = await supabase.from('users_profile').select('*').order('id', { ascending: true })
    if (uData) setUsersList(uData)

    const { data: gData } = await supabase.from('games').select(`*, creator:users_profile(nickname)`).order('created_at', { ascending: false })
    if (gData) setGamesList(gData)

    const { data: rData } = await supabase.from('recruits').select(`*, creator:users_profile(nickname), games(name)`).order('created_at', { ascending: false })
    if (rData) setRecruitsList(rData)
  }

  async function fetchLogs() {
    const { data, error } = await supabase
      .from('admin_logs')
      .select(`*, admin:users_profile(nickname)`)
      .order('created_at', { ascending: false })
      .limit(100)
    if (error) return setMsg('璇诲彇鏃ュ織澶辫触锛? + error.message)
    setLogsList(data || [])
  }

  // 缁熶竴鎵ц RPC锛屽鐞嗛敊璇彁绀?
  async function run(fn: () => Promise<any>, okText: string) {
    setBusy(true)
    setMsg('')
    const { error } = await fn()
    setBusy(false)
    if (error) {
      setMsg(error.message)
      return false
    }
    setMsg(okText)
    await fetchData()
    return true
  }

  // 鏀圭粡楠岋細鐣岄潰涓婅緭鍏ョ粷瀵瑰€硷紝鎸夊樊鍊艰蛋 RPC
  async function handleUpdateExp(u: any) {
    const currentExp = u.experience || 0
    const input = prompt(`涓?[ID:${u.id} ${u.nickname}] 璁剧疆鏂扮殑缁忛獙鍊硷細`, currentExp.toString())
    if (input === null) return
    const target = parseInt(input)
    if (isNaN(target) || target < 0) return alert('璇疯緭鍏ュ悎娉曠殑闈炶礋鏁存暟')

    const delta = target - currentExp
    if (delta === 0) return

    const reason = prompt('鎿嶄綔鍘熷洜锛堝彲鐣欑┖锛屼細鍐欒繘鏃ュ織锛?) || null
    await run(
      () => supabase.rpc('admin_add_experience', { p_target_id: u.id, p_delta: delta, p_reason: reason }),
      `宸叉妸 [ID:${u.id}] 鐨勭粡楠屼粠 ${currentExp} 璋冩暣涓?${target}`
    )
  }

  // 灏佺 / 瑙ｅ皝
  async function handleBanUser(u: any) {
    const isBanned = u.banned_until && new Date(u.banned_until) > new Date()

    if (isBanned) {
      if (!confirm(`瑙ｉ櫎 [ID:${u.id} ${u.nickname}] 鐨勫皝绂侊紵`)) return
      await run(
        () => supabase.rpc('admin_unban_user', { p_target_id: u.id }),
        `宸茶В灏?${u.nickname}`
      )
      return
    }

    const days = prompt('璇疯緭鍏ュ皝绂佸ぉ鏁帮紙1 / 3 / 7 / 30 / 365锛夛細')
    if (days === null) return
    const numDays = parseInt(days)
    if (isNaN(numDays) || numDays <= 0) return alert('澶╂暟涓嶅悎娉?)

    const reason = prompt('灏佺鍘熷洜锛堝彲鐣欑┖锛屼細鍐欒繘鏃ュ織锛?) || null
    await run(
      () => supabase.rpc('admin_ban_user', { p_target_id: u.id, p_days: numDays, p_reason: reason }),
      `宸插皝绂?${u.nickname} ${numDays} 澶ー
    )
  }

  // 璁捐鑹诧紙鍙湁瓒呯骇绠＄悊鍛樿兘鐪嬪埌鎸夐挳锛?
  async function handleSetRole(u: any) {
    const input = prompt(
      `璁剧疆 [ID:${u.id} ${u.nickname}] 鐨勮鑹诧細\nuser = 鏅€氱帺瀹禱nadmin = 绠＄悊鍛榎nsuper_admin = 瓒呯骇绠＄悊鍛榒,
      u.role || 'user'
    )
    if (input === null) return

    const role = input.trim()
    if (!['user', 'admin', 'super_admin'].includes(role)) {
      return alert('鍙兘濉?user銆乤dmin 鎴?super_admin')
    }
    if (!confirm(`纭鎶?${u.nickname} 璁句负 ${role}锛焋)) return

    await run(
      () => supabase.rpc('admin_set_role', { p_target_id: u.id, p_role: role }),
      `宸叉妸 ${u.nickname} 璁句负 ${role}`
    )
  }

  async function handleDeleteGame(gameId: number, name: string) {
    if (!confirm(`鈿狅笍 纭畾瑕佸垹闄ゃ€?{name}銆嶅強鎵€鏈夌浉鍏虫暟鎹悧锛熸鎿嶄綔涓嶅彲鎾ら攢銆俙)) return
    await run(
      () => supabase.rpc('admin_delete_content', { p_table: 'games', p_id: gameId }),
      `宸插垹闄ゆ父鎴忋€?{name}銆峘
    )
  }

  async function handleDeleteRecruit(recruitId: number) {
    if (!confirm('鈿狅笍 纭畾瑕佹竻闄よ繖鏉¤繚瑙勬偓璧忎护鍚楋紵')) return
    await run(
      () => supabase.rpc('admin_delete_content', { p_table: 'recruits', p_id: recruitId }),
      '宸插垹闄よ鏉￠€氱級浠?
    )
  }

  if (loading) return <div className="min-h-screen bg-[#050B14] flex items-center justify-center text-cyan-500 font-orbitron animate-pulse">楠岃瘉绁炴潈韬唤涓?..</div>
  if (!isAdmin) return null

  const isSuper = userProfile?.role === 'super_admin'

  const filteredUsers = usersList.filter(u =>
    u.nickname?.toLowerCase().includes(userSearch.toLowerCase()) ||
    u.id.toString().includes(userSearch) ||
    (u.email || '').toLowerCase().includes(userSearch.toLowerCase())
  )
  const filteredGames = gamesList.filter(g => (g.name || '').toLowerCase().includes(gameSearch.toLowerCase()))
  const filteredRecruits = recruitsList.filter(r =>
    (r.play_content || '').toLowerCase().includes(recruitSearch.toLowerCase()) ||
    (r.games?.name || '').toLowerCase().includes(recruitSearch.toLowerCase())
  )

  return (
    <div className="min-h-screen bg-[#050B14] text-gray-200 flex overflow-hidden">

      {/* 渚ц竟鏍?*/}
      <div className="w-64 bg-[#0A111F] border-r border-cyan-900/50 flex flex-col z-20 shadow-[10px_0_30px_rgba(0,0,0,0.5)]">
        <div className="p-6 border-b border-gray-800">
          <h1 className="text-xl font-black text-transparent bg-clip-text bg-gradient-to-r from-fuchsia-500 to-red-500 font-orbitron tracking-widest">
            绁炴潈鎺у埗鍙?
          </h1>
          <p className="text-xs text-gray-500 mt-2">
            宸℃父澶╁畼: {userProfile.nickname}
          </p>
          <p className="text-[10px] text-red-400 mt-1 font-mono">
            {isSuper ? '鏉冮檺绾у埆锛氳秴绾х鐞嗗憳' : '鏉冮檺绾у埆锛氱鐞嗗憳'}
          </p>
        </div>

        <nav className="flex-1 p-4 space-y-2">
          <button onClick={() => setActiveTab('overview')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'overview' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>馃搳 鏁版嵁鎬昏</button>
          <button onClick={() => setActiveTab('users')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'users' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>馃懃 鐢ㄦ埛绠＄悊</button>
          <button onClick={() => setActiveTab('games')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'games' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>馃幃 娓告垙妯″潡</button>
          <button onClick={() => setActiveTab('recruits')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'recruits' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>馃毃 鎮祻閫氱級</button>
          <button onClick={() => { setActiveTab('logs'); fetchLogs() }} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'logs' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>馃摐 鎿嶄綔鏃ュ織</button>
        </nav>

        <div className="p-4 border-t border-gray-800">
          <Link href="/" className="block w-full text-center bg-gray-800 text-gray-300 py-3 rounded-xl font-bold hover:bg-gray-700 transition">杩斿洖鍓嶅彴</Link>
        </div>
      </div>

      {/* 涓诲唴瀹瑰尯 */}
      <div className="flex-1 overflow-y-auto p-8 relative">
        <div className="absolute top-0 right-0 w-[40vw] h-[40vw] bg-fuchsia-600/5 rounded-full blur-[100px] pointer-events-none"></div>

        {msg && (
          <div className="relative z-20 mb-6 text-sm text-cyan-300 bg-cyan-900/20 border border-cyan-500/30 rounded-xl py-2.5 px-4 flex items-center justify-between">
            <span>{msg}</span>
            <button onClick={() => setMsg('')} className="text-gray-500 hover:text-white ml-4">鉁?/button>
          </div>
        )}

        {/* 1. 鏁版嵁鎬昏 */}
        {activeTab === 'overview' && (
          <div className="space-y-6 relative z-10">
            <h2 className="text-3xl font-black text-white font-orbitron mb-8">绯荤粺鏁版嵁鎬昏</h2>
            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-4 gap-6">
              <div className="bg-[#0A111F] border border-cyan-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-cyan-500 font-mono mb-2">// TOTAL USERS</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.users}</p>
              </div>
              <div className="bg-[#0A111F] border border-blue-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-blue-500 font-mono mb-2">// TOTAL GAMES</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.games}</p>
              </div>
              <div className="bg-[#0A111F] border border-fuchsia-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-fuchsia-500 font-mono mb-2">// TOTAL RECRUITS</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.recruits}</p>
              </div>
              <div className="bg-[#0A111F] border border-yellow-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-yellow-500 font-mono mb-2">// TOTAL MOMENTS</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.moments}</p>
              </div>
            </div>
          </div>
        )}

        {/* 2. 鐢ㄦ埛绠＄悊 */}
        {activeTab === 'users' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8 flex-wrap gap-4">
              <h2 className="text-3xl font-black text-white font-orbitron">鐢ㄦ埛鏉冮檺涓庡皝绂?/h2>
              <input
                type="text"
                placeholder="鎼滅储 ID / 鏄电О / 閭..."
                value={userSearch}
                onChange={e => setUserSearch(e.target.value)}
                className="bg-[#0A111F] border border-cyan-900/50 text-white px-4 py-2 rounded-xl outline-none focus:border-cyan-500 w-64"
              />
            </div>

            <div className="bg-[#0A111F] border border-gray-800 rounded-2xl overflow-hidden overflow-x-auto">
              <table className="w-full text-left text-sm min-w-[760px]">
                <thead className="bg-black/50 text-cyan-400 font-mono">
                  <tr>
                    <th className="p-4">ID</th>
                    <th className="p-4">鐢ㄦ埛</th>
                    <th className="p-4">绛夌骇</th>
                    <th className="p-4">缁忛獙</th>
                    <th className="p-4">鐘舵€?/th>
                    <th className="p-4">鎿嶄綔</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-800">
                  {filteredUsers.map(u => {
                    const isBanned = u.banned_until && new Date(u.banned_until) > new Date()
                    const lvl = calculateLevel(u.experience || 0, u.id)
                    const isSelf = u.id === userProfile.id
                    return (
                      <tr key={u.id} className="hover:bg-white/5 transition">
                        <td className="p-4 font-mono font-bold">{u.id}</td>
                        <td className="p-4">
                          <div className="flex items-center gap-2">
                            <img src={u.avatar_url || '/logo.png'} className="w-6 h-6 rounded-full object-cover bg-black" />
                            <span className="truncate max-w-[140px]">{u.nickname}</span>
                            {u.role === 'super_admin' && <span className="text-[9px] bg-red-900/50 text-red-300 border border-red-500/40 px-1.5 py-0.5 rounded shrink-0">瓒呯</span>}
                            {u.role === 'admin' && <span className="text-[9px] bg-fuchsia-900/50 text-fuchsia-300 border border-fuchsia-500/40 px-1.5 py-0.5 rounded shrink-0">绠＄悊</span>}
                            {isSelf && <span className="text-[9px] text-cyan-500 shrink-0">(浣?</span>}
                          </div>
                        </td>
                        <td className="p-4 text-gray-400 font-mono text-xs">Lv.{lvl.level}</td>
                        <td className="p-4 text-cyan-400 font-bold">{u.experience || 0}</td>
                        <td className="p-4">
                          {isBanned
                            ? <span className="text-red-500 font-bold text-xs">宸插皝绂?/span>
                            : <span className="text-green-500 text-xs">姝ｅ父</span>}
                        </td>
                        <td className="p-4">
                          <div className="flex gap-2 flex-wrap">
                            <button
                              onClick={() => handleUpdateExp(u)}
                              disabled={busy}
                              className="text-xs bg-gray-800 border border-gray-600 px-3 py-1.5 rounded hover:bg-gray-700 disabled:opacity-40 whitespace-nowrap"
                            >
                              鏀圭粡楠?
                            </button>
                            <button
                              onClick={() => handleBanUser(u)}
                              disabled={busy || u.role === 'super_admin' || isSelf}
                              title={u.role === 'super_admin' ? '涓嶈兘灏佺瓒呯骇绠＄悊鍛? : isSelf ? '涓嶈兘灏佺鑷繁' : ''}
                              className={`text-xs border px-3 py-1.5 rounded disabled:opacity-30 disabled:cursor-not-allowed whitespace-nowrap ${isBanned ? 'bg-green-900/20 text-green-400 border-green-600' : 'bg-red-900/20 text-red-500 border-red-600'}`}
                            >
                              {isBanned ? '瑙ｅ皝' : '灏佺'}
                            </button>
                            {isSuper && (
                              <button
                                onClick={() => handleSetRole(u)}
                                disabled={busy}
                                className="text-xs bg-fuchsia-900/20 text-fuchsia-400 border border-fuchsia-600 px-3 py-1.5 rounded hover:bg-fuchsia-900/40 disabled:opacity-40 whitespace-nowrap"
                              >
                                璁捐鑹?
                              </button>
                            )}
                          </div>
                        </td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
              {filteredUsers.length === 0 && <p className="p-6 text-center text-gray-500">娌℃湁鎵惧埌鐢ㄦ埛</p>}
            </div>
          </div>
        )}

        {/* 3. 娓告垙妯″潡绠＄悊 */}
        {activeTab === 'games' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8 flex-wrap gap-4">
              <h2 className="text-3xl font-black text-white font-orbitron">娓告垙妯″潡娓呯悊</h2>
              <input
                type="text"
                placeholder="鎼滅储娓告垙鍚?.."
                value={gameSearch}
                onChange={e => setGameSearch(e.target.value)}
                className="bg-[#0A111F] border border-cyan-900/50 text-white px-4 py-2 rounded-xl outline-none focus:border-cyan-500 w-64"
              />
            </div>
            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
              {filteredGames.map(g => (
                <div key={g.id} className="bg-[#0A111F] border border-gray-800 p-4 rounded-xl flex items-center justify-between gap-3">
                  <div className="flex items-center gap-3 min-w-0">
                    <img src={g.cover_url || '/logo.png'} className="w-10 h-10 rounded object-cover shrink-0" />
                    <div className="min-w-0">
                      <p className="font-bold text-white truncate">{g.name}</p>
                      <p className="text-xs text-gray-500 truncate">鍒涘缓鑰? {g.creator?.nickname || '鏈煡'}</p>
                    </div>
                  </div>
                  <button
                    onClick={() => handleDeleteGame(g.id, g.name)}
                    disabled={busy}
                    className="text-red-500 bg-red-900/20 p-2 rounded hover:bg-red-900/50 shrink-0 disabled:opacity-40"
                  >
                    馃棏锔?
                  </button>
                </div>
              ))}
              {filteredGames.length === 0 && <p className="col-span-full text-gray-500">娌℃湁鎵惧埌妯″潡</p>}
            </div>
          </div>
        )}

        {/* 4. 鎮祻浠ょ鐞?*/}
        {activeTab === 'recruits' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8 flex-wrap gap-4">
              <h2 className="text-3xl font-black text-white font-orbitron">杩濊鎮祻娓呯悊</h2>
              <input
                type="text"
                placeholder="鎼滅储鎮祻鍐呭 / 娓告垙鍚?.."
                value={recruitSearch}
                onChange={e => setRecruitSearch(e.target.value)}
                className="bg-[#0A111F] border border-cyan-900/50 text-white px-4 py-2 rounded-xl outline-none focus:border-cyan-500 w-64"
              />
            </div>
            <div className="space-y-3">
              {filteredRecruits.map(r => (
                <div key={r.id} className="bg-[#0A111F] border border-gray-800 p-4 rounded-xl flex items-center justify-between gap-4">
                  <div className="min-w-0">
                    <p className="font-bold text-white truncate">{r.play_content}</p>
                    <p className="text-xs text-gray-500 mt-1">娓告垙: {r.games?.name || '鏈煡'} | 鍙戝竷鑰? {r.creator?.nickname || '鏈煡'} | 鏃堕棿: {r.online_time}</p>
                  </div>
                  <button
                    onClick={() => handleDeleteRecruit(r.id)}
                    disabled={busy}
                    className="text-red-500 bg-red-900/20 px-3 py-2 rounded hover:bg-red-900/50 shrink-0 text-xs font-bold disabled:opacity-40"
                  >
                    馃棏锔?鍒犻櫎
                  </button>
                </div>
              ))}
              {filteredRecruits.length === 0 && <p className="text-gray-500">娌℃湁鎵惧埌閫氱級浠?/p>}
            </div>

        if (followingData && followingData.length > 0) {

          if (allFollowingData) setFollowing(allFollowingData)
        }
      }

    init()

    return () => window.removeEventListener('openChatWidget', handleOpen)

    if (activeContact && userProfile && isOpen) {
      fetchMessages(activeContact.id, false)

    }
    return () => { if (interval) clearInterval(interval) }
  }, [activeContact, userProfile, isOpen])

      .or(`and(sender_id.eq.${userProfile.id},receiver_id.eq.${contactId}),and(sender_id.eq.${contactId},receiver_id.eq.${userProfile.id})`)

import { useState, useEffect } from 'react'
import { supabase } from '@/lib/supabase'

export default function FollowUserButton({ targetId, currentUserId }: { targetId: number, currentUserId?: number }) {
  const [isFollowing, setIsFollowing] = useState(false)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!currentUserId || currentUserId === targetId) { setLoading(false); return }
    supabase.from('follows').select('id').eq('follower_id', currentUserId).eq('following_id', targetId).single().then(({ data }) => {
      setIsFollowing(!!data)
      setLoading(false)
    })
  }, [currentUserId, targetId])

  async function toggleFollow(e: React.MouseEvent) {
    e.preventDefault()
    e.stopPropagation()
    if (!currentUserId) return alert('璇峰厛鐧诲綍锛?)
    if (currentUserId === targetId) return

    if (isFollowing) {
      await supabase.from('follows').delete().eq('follower_id', currentUserId).eq('following_id', targetId)
      setIsFollowing(false)
    } else {
      await supabase.from('follows').insert({ follower_id: currentUserId, following_id: targetId })
      await supabase.from('notifications').insert({ user_id: targetId, actor_id: currentUserId, type: 'follow' })
      setIsFollowing(true)
    }
  }

  if (loading || !currentUserId || currentUserId === targetId) return null

  return (
    <button 
      onClick={toggleFollow} 
      className={`px-3 py-1 rounded text-xs font-bold transition-all border ${
        isFollowing 
          ? 'bg-cyan-900/40 text-cyan-400 border-cyan-500/50' 
          : 'bg-transparent text-blue-400 border-blue-500/50 hover:bg-blue-500/20'
      }`}
    >
      {isFollowing ? '宸插叧娉? : '+ 鍏虫敞'}
    </button>
  )
}

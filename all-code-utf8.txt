===== app\admin\page.tsx =====
'use client'

import { useState, useEffect } from 'react'
import { supabase } from '@/lib/supabase'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { calculateLevel } from '@/lib/levels'

type Tab = 'overview' | 'users' | 'games' | 'recruits' | 'logs'

export default function AdminPage() {
  const router = useRouter()
  const [userProfile, setUserProfile] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  const [isAdmin, setIsAdmin] = useState(false)
  const [busy, setBusy] = useState(false)
  const [msg, setMsg] = useState('')

  const [activeTab, setActiveTab] = useState<Tab>('overview')

  const [stats, setStats] = useState({ users: 0, games: 0, recruits: 0, moments: 0 })
  const [usersList, setUsersList] = useState<any[]>([])
  const [gamesList, setGamesList] = useState<any[]>([])
  const [recruitsList, setRecruitsList] = useState<any[]>([])
  const [logsList, setLogsList] = useState<any[]>([])

  const [userSearch, setUserSearch] = useState('')
  const [gameSearch, setGameSearch] = useState('')
  const [recruitSearch, setRecruitSearch] = useState('')

  useEffect(() => {
    async function checkAuth() {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) {
        alert('未登录，无法访问后台！')
        router.push('/login')
        return
      }

      const { data: profile } = await supabase.from('users_profile').select('*').eq('auth_id', user.id).single()

      if (!profile || !['admin', 'super_admin'].includes(profile.role)) {
        alert('⛔ 权限不足：你的账号没有后台权限')
        router.push('/')
        return
      }

      setUserProfile(profile)
      setIsAdmin(true)
      await fetchData()
      setLoading(false)
    }
    checkAuth()
  }, [router])

  async function fetchData() {
    const { count: userCount } = await supabase.from('users_profile').select('*', { count: 'exact', head: true })
    const { count: gameCount } = await supabase.from('games').select('*', { count: 'exact', head: true })
    const { count: recruitCount } = await supabase.from('recruits').select('*', { count: 'exact', head: true })
    const { count: momentCount } = await supabase.from('moments').select('*', { count: 'exact', head: true })
    setStats({
      users: userCount || 0,
      games: gameCount || 0,
      recruits: recruitCount || 0,
      moments: momentCount || 0,
    })

    const { data: uData } = await supabase.from('users_profile').select('*').order('id', { ascending: true })
    if (uData) setUsersList(uData)

    const { data: gData } = await supabase.from('games').select(`*, creator:users_profile(nickname)`).order('created_at', { ascending: false })
    if (gData) setGamesList(gData)

    const { data: rData } = await supabase.from('recruits').select(`*, creator:users_profile(nickname), games(name)`).order('created_at', { ascending: false })
    if (rData) setRecruitsList(rData)
  }

  async function fetchLogs() {
    const { data, error } = await supabase
      .from('admin_logs')
      .select(`*, admin:users_profile(nickname)`)
      .order('created_at', { ascending: false })
      .limit(100)
    if (error) return setMsg('读取日志失败：' + error.message)
    setLogsList(data || [])
  }

  // 统一执行 RPC，处理错误提示
  async function run(fn: () => Promise<any>, okText: string) {
    setBusy(true)
    setMsg('')
    const { error } = await fn()
    setBusy(false)
    if (error) {
      setMsg(error.message)
      return false
    }
    setMsg(okText)
    await fetchData()
    return true
  }

  // 改经验：界面上输入绝对值，按差值走 RPC
  async function handleUpdateExp(u: any) {
    const currentExp = u.experience || 0
    const input = prompt(`为 [ID:${u.id} ${u.nickname}] 设置新的经验值：`, currentExp.toString())
    if (input === null) return
    const target = parseInt(input)
    if (isNaN(target) || target < 0) return alert('请输入合法的非负整数')

    const delta = target - currentExp
    if (delta === 0) return

    const reason = prompt('操作原因（可留空，会写进日志）') || null
    await run(
      () => supabase.rpc('admin_add_experience', { p_target_id: u.id, p_delta: delta, p_reason: reason }),
      `已把 [ID:${u.id}] 的经验从 ${currentExp} 调整为 ${target}`
    )
  }

  // 封禁 / 解封
  async function handleBanUser(u: any) {
    const isBanned = u.banned_until && new Date(u.banned_until) > new Date()

    if (isBanned) {
      if (!confirm(`解除 [ID:${u.id} ${u.nickname}] 的封禁？`)) return
      await run(
        () => supabase.rpc('admin_unban_user', { p_target_id: u.id }),
        `已解封 ${u.nickname}`
      )
      return
    }

    const days = prompt('请输入封禁天数（1 / 3 / 7 / 30 / 365）：')
    if (days === null) return
    const numDays = parseInt(days)
    if (isNaN(numDays) || numDays <= 0) return alert('天数不合法')

    const reason = prompt('封禁原因（可留空，会写进日志）') || null
    await run(
      () => supabase.rpc('admin_ban_user', { p_target_id: u.id, p_days: numDays, p_reason: reason }),
      `已封禁 ${u.nickname} ${numDays} 天`
    )
  }

  // 设角色（只有超级管理员能看到按钮）
  async function handleSetRole(u: any) {
    const input = prompt(
      `设置 [ID:${u.id} ${u.nickname}] 的角色：\nuser = 普通玩家\nadmin = 管理员\nsuper_admin = 超级管理员`,
      u.role || 'user'
    )
    if (input === null) return

    const role = input.trim()
    if (!['user', 'admin', 'super_admin'].includes(role)) {
      return alert('只能填 user、admin 或 super_admin')
    }
    if (!confirm(`确认把 ${u.nickname} 设为 ${role}？`)) return

    await run(
      () => supabase.rpc('admin_set_role', { p_target_id: u.id, p_role: role }),
      `已把 ${u.nickname} 设为 ${role}`
    )
  }

  async function handleDeleteGame(gameId: number, name: string) {
    if (!confirm(`⚠️ 确定要删除「${name}」及所有相关数据吗？此操作不可撤销。`)) return
    await run(
      () => supabase.rpc('admin_delete_content', { p_table: 'games', p_id: gameId }),
      `已删除游戏「${name}」`
    )
  }

  async function handleDeleteRecruit(recruitId: number) {
    if (!confirm('⚠️ 确定要清除这条违规悬赏令吗？')) return
    await run(
      () => supabase.rpc('admin_delete_content', { p_table: 'recruits', p_id: recruitId }),
      '已删除该条通缉令'
    )
  }

  if (loading) return <div className="min-h-screen bg-[#050B14] flex items-center justify-center text-cyan-500 font-orbitron animate-pulse">验证神权身份中...</div>
  if (!isAdmin) return null

  const isSuper = userProfile?.role === 'super_admin'

  const filteredUsers = usersList.filter(u =>
    u.nickname?.toLowerCase().includes(userSearch.toLowerCase()) ||
    u.id.toString().includes(userSearch) ||
    (u.email || '').toLowerCase().includes(userSearch.toLowerCase())
  )
  const filteredGames = gamesList.filter(g => (g.name || '').toLowerCase().includes(gameSearch.toLowerCase()))
  const filteredRecruits = recruitsList.filter(r =>
    (r.play_content || '').toLowerCase().includes(recruitSearch.toLowerCase()) ||
    (r.games?.name || '').toLowerCase().includes(recruitSearch.toLowerCase())
  )

  return (
    <div className="min-h-screen bg-[#050B14] text-gray-200 flex overflow-hidden">

      {/* 侧边栏 */}
      <div className="w-64 bg-[#0A111F] border-r border-cyan-900/50 flex flex-col z-20 shadow-[10px_0_30px_rgba(0,0,0,0.5)]">
        <div className="p-6 border-b border-gray-800">
          <h1 className="text-xl font-black text-transparent bg-clip-text bg-gradient-to-r from-fuchsia-500 to-red-500 font-orbitron tracking-widest">
            神权控制台
          </h1>
          <p className="text-xs text-gray-500 mt-2">
            巡游天官: {userProfile.nickname}
          </p>
          <p className="text-[10px] text-red-400 mt-1 font-mono">
            {isSuper ? '权限级别：超级管理员' : '权限级别：管理员'}
          </p>
        </div>

        <nav className="flex-1 p-4 space-y-2">
          <button onClick={() => setActiveTab('overview')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'overview' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>📊 数据总览</button>
          <button onClick={() => setActiveTab('users')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'users' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>👥 用户管理</button>
          <button onClick={() => setActiveTab('games')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'games' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>🎮 游戏模块</button>
          <button onClick={() => setActiveTab('recruits')} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'recruits' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>🚨 悬赏通缉</button>
          <button onClick={() => { setActiveTab('logs'); fetchLogs() }} className={`w-full text-left px-4 py-3 rounded-xl font-bold transition-all ${activeTab === 'logs' ? 'bg-cyan-600 text-white shadow-[0_0_15px_rgba(6,182,212,0.5)]' : 'text-gray-400 hover:bg-gray-800'}`}>📜 操作日志</button>
        </nav>

        <div className="p-4 border-t border-gray-800">
          <Link href="/" className="block w-full text-center bg-gray-800 text-gray-300 py-3 rounded-xl font-bold hover:bg-gray-700 transition">返回前台</Link>
        </div>
      </div>

      {/* 主内容区 */}
      <div className="flex-1 overflow-y-auto p-8 relative">
        <div className="absolute top-0 right-0 w-[40vw] h-[40vw] bg-fuchsia-600/5 rounded-full blur-[100px] pointer-events-none"></div>

        {msg && (
          <div className="relative z-20 mb-6 text-sm text-cyan-300 bg-cyan-900/20 border border-cyan-500/30 rounded-xl py-2.5 px-4 flex items-center justify-between">
            <span>{msg}</span>
            <button onClick={() => setMsg('')} className="text-gray-500 hover:text-white ml-4">✕</button>
          </div>
        )}

        {/* 1. 数据总览 */}
        {activeTab === 'overview' && (
          <div className="space-y-6 relative z-10">
            <h2 className="text-3xl font-black text-white font-orbitron mb-8">系统数据总览</h2>
            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-4 gap-6">
              <div className="bg-[#0A111F] border border-cyan-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-cyan-500 font-mono mb-2">// TOTAL USERS</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.users}</p>
              </div>
              <div className="bg-[#0A111F] border border-blue-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-blue-500 font-mono mb-2">// TOTAL GAMES</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.games}</p>
              </div>
              <div className="bg-[#0A111F] border border-fuchsia-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-fuchsia-500 font-mono mb-2">// TOTAL RECRUITS</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.recruits}</p>
              </div>
              <div className="bg-[#0A111F] border border-yellow-500/30 p-6 rounded-2xl shadow-lg">
                <p className="text-yellow-500 font-mono mb-2">// TOTAL MOMENTS</p>
                <p className="text-5xl font-black text-white font-orbitron">{stats.moments}</p>
              </div>
            </div>
          </div>
        )}

        {/* 2. 用户管理 */}
        {activeTab === 'users' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8 flex-wrap gap-4">
              <h2 className="text-3xl font-black text-white font-orbitron">用户权限与封禁</h2>
              <input
                type="text"
                placeholder="搜索 ID / 昵称 / 邮箱..."
                value={userSearch}
                onChange={e => setUserSearch(e.target.value)}
                className="bg-[#0A111F] border border-cyan-900/50 text-white px-4 py-2 rounded-xl outline-none focus:border-cyan-500 w-64"
              />
            </div>

            <div className="bg-[#0A111F] border border-gray-800 rounded-2xl overflow-hidden overflow-x-auto">
              <table className="w-full text-left text-sm min-w-[760px]">
                <thead className="bg-black/50 text-cyan-400 font-mono">
                  <tr>
                    <th className="p-4">ID</th>
                    <th className="p-4">用户</th>
                    <th className="p-4">等级</th>
                    <th className="p-4">经验</th>
                    <th className="p-4">状态</th>
                    <th className="p-4">操作</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-800">
                  {filteredUsers.map(u => {
                    const isBanned = u.banned_until && new Date(u.banned_until) > new Date()
                    const lvl = calculateLevel(u.experience || 0, u.id)
                    const isSelf = u.id === userProfile.id
                    return (
                      <tr key={u.id} className="hover:bg-white/5 transition">
                        <td className="p-4 font-mono font-bold">{u.id}</td>
                        <td className="p-4">
                          <div className="flex items-center gap-2">
                            <img src={u.avatar_url || '/logo.png'} className="w-6 h-6 rounded-full object-cover bg-black" />
                            <span className="truncate max-w-[140px]">{u.nickname}</span>
                            {u.role === 'super_admin' && <span className="text-[9px] bg-red-900/50 text-red-300 border border-red-500/40 px-1.5 py-0.5 rounded shrink-0">超管</span>}
                            {u.role === 'admin' && <span className="text-[9px] bg-fuchsia-900/50 text-fuchsia-300 border border-fuchsia-500/40 px-1.5 py-0.5 rounded shrink-0">管理</span>}
                            {isSelf && <span className="text-[9px] text-cyan-500 shrink-0">(你)</span>}
                          </div>
                        </td>
                        <td className="p-4 text-gray-400 font-mono text-xs">Lv.{lvl.level}</td>
                        <td className="p-4 text-cyan-400 font-bold">{u.experience || 0}</td>
                        <td className="p-4">
                          {isBanned
                            ? <span className="text-red-500 font-bold text-xs">已封禁</span>
                            : <span className="text-green-500 text-xs">正常</span>}
                        </td>
                        <td className="p-4">
                          <div className="flex gap-2 flex-wrap">
                            <button
                              onClick={() => handleUpdateExp(u)}
                              disabled={busy}
                              className="text-xs bg-gray-800 border border-gray-600 px-3 py-1.5 rounded hover:bg-gray-700 disabled:opacity-40 whitespace-nowrap"
                            >
                              改经验
                            </button>
                            <button
                              onClick={() => handleBanUser(u)}
                              disabled={busy || u.role === 'super_admin' || isSelf}
                              title={u.role === 'super_admin' ? '不能封禁超级管理员' : isSelf ? '不能封禁自己' : ''}
                              className={`text-xs border px-3 py-1.5 rounded disabled:opacity-30 disabled:cursor-not-allowed whitespace-nowrap ${isBanned ? 'bg-green-900/20 text-green-400 border-green-600' : 'bg-red-900/20 text-red-500 border-red-600'}`}
                            >
                              {isBanned ? '解封' : '封禁'}
                            </button>
                            {isSuper && (
                              <button
                                onClick={() => handleSetRole(u)}
                                disabled={busy}
                                className="text-xs bg-fuchsia-900/20 text-fuchsia-400 border border-fuchsia-600 px-3 py-1.5 rounded hover:bg-fuchsia-900/40 disabled:opacity-40 whitespace-nowrap"
                              >
                                设角色
                              </button>
                            )}
                          </div>
                        </td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
              {filteredUsers.length === 0 && <p className="p-6 text-center text-gray-500">没有找到用户</p>}
            </div>
          </div>
        )}

        {/* 3. 游戏模块管理 */}
        {activeTab === 'games' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8 flex-wrap gap-4">
              <h2 className="text-3xl font-black text-white font-orbitron">游戏模块清理</h2>
              <input
                type="text"
                placeholder="搜索游戏名..."
                value={gameSearch}
                onChange={e => setGameSearch(e.target.value)}
                className="bg-[#0A111F] border border-cyan-900/50 text-white px-4 py-2 rounded-xl outline-none focus:border-cyan-500 w-64"
              />
            </div>
            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
              {filteredGames.map(g => (
                <div key={g.id} className="bg-[#0A111F] border border-gray-800 p-4 rounded-xl flex items-center justify-between gap-3">
                  <div className="flex items-center gap-3 min-w-0">
                    <img src={g.cover_url || '/logo.png'} className="w-10 h-10 rounded object-cover shrink-0" />
                    <div className="min-w-0">
                      <p className="font-bold text-white truncate">{g.name}</p>
                      <p className="text-xs text-gray-500 truncate">创建者: {g.creator?.nickname || '未知'}</p>
                    </div>
                  </div>
                  <button
                    onClick={() => handleDeleteGame(g.id, g.name)}
                    disabled={busy}
                    className="text-red-500 bg-red-900/20 p-2 rounded hover:bg-red-900/50 shrink-0 disabled:opacity-40"
                  >
                    🗑️
                  </button>
                </div>
              ))}
              {filteredGames.length === 0 && <p className="col-span-full text-gray-500">没有找到模块</p>}
            </div>
          </div>
        )}

        {/* 4. 悬赏令管理 */}
        {activeTab === 'recruits' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8 flex-wrap gap-4">
              <h2 className="text-3xl font-black text-white font-orbitron">违规悬赏清理</h2>
              <input
                type="text"
                placeholder="搜索悬赏内容 / 游戏名..."
                value={recruitSearch}
                onChange={e => setRecruitSearch(e.target.value)}
                className="bg-[#0A111F] border border-cyan-900/50 text-white px-4 py-2 rounded-xl outline-none focus:border-cyan-500 w-64"
              />
            </div>
            <div className="space-y-3">
              {filteredRecruits.map(r => (
                <div key={r.id} className="bg-[#0A111F] border border-gray-800 p-4 rounded-xl flex items-center justify-between gap-4">
                  <div className="min-w-0">
                    <p className="font-bold text-white truncate">{r.play_content}</p>
                    <p className="text-xs text-gray-500 mt-1">游戏: {r.games?.name || '未知'} | 发布者: {r.creator?.nickname || '未知'} | 时间: {r.online_time}</p>
                  </div>
                  <button
                    onClick={() => handleDeleteRecruit(r.id)}
                    disabled={busy}
                    className="text-red-500 bg-red-900/20 px-3 py-2 rounded hover:bg-red-900/50 shrink-0 text-xs font-bold disabled:opacity-40"
                  >
                    🗑️ 删除
                  </button>
                </div>
              ))}
              {filteredRecruits.length === 0 && <p className="text-gray-500">没有找到通缉令</p>}
            </div>
          </div>
        )}

        {/* 5. 操作日志 */}
        {activeTab === 'logs' && (
          <div className="relative z-10">
            <div className="flex justify-between items-center mb-8">
              <h2 className="text-3xl font-black text-white font-orbitron">操作日志</h2>
              <button
                onClick={fetchLogs}
                className="text-xs bg-gray-800 border border-gray-600 px-4 py-2 rounded-xl hover:bg-gray-700"
              >
                🔄 刷新
              </button>
            </div>
            <div className="space-y-2">
              {logsList.map(l => (
                <div key={l.id} className="bg-[#0A111F] border border-gray-800 rounded-xl p-3 text-xs font-mono flex flex-wrap items-center gap-x-2 gap-y-1">
                  <span className="text-cyan-400">{new Date(l.created_at).toLocaleString('zh-CN')}</span>
                  <span className="text-gray-600">|</span>
                  <span className="text-white">{l.admin?.nickname || `ID:${l.admin_id}`}</span>
                  <span className="text-gray-600">→</span>
                  <span className="text-fuchsia-400">{l.action}</span>
                  <span className="text-gray-500">{l.target_type}:{l.target_id}</span>
                  {l.detail && <span className="text-gray-600">{JSON.stringify(l.detail)}</span>}
                </div>
              ))}
              {logsList.length === 0 && (
                <p className="text-gray-500 text-center py-12">暂无操作记录</p>
              )}
            </div>
          </div>
        )}

      </div>
    </div>
  )
}

import { NextResponse } from 'next/server'
import { Resend } from 'resend'

const resend = new Resend(process.env.RESEND_API_KEY)

export async function POST(req: Request) {
  try {
    const { type, content, contact } = await req.json()

    const data = await resend.emails.send({
      from: 'Wingman反馈 <noreply@1288688.xyz>', // 必须是你在 Resend 验证过的域名
      to: ['ydhbmb@163.com'], // 🔥 改成你接收反馈的邮箱
      subject: `【Wingman反馈】收到一条新的${type}`,
      html: `
        <div style="font-family: sans-serif; padding: 20px; background: #f4f4f4;">
          <h2 style="color: #0ea5e9;">Wingman Games 收到新反馈</h2>
          <p><strong>反馈类型：</strong>${type}</p>
          <p><strong>联系方式：</strong>${contact || '未留'}</p>
          <p><strong>具体内容：</strong></p>
          <div style="background: #fff; padding: 15px; border-radius: 8px; border: 1px solid #ddd;">
            ${content}
          </div>
        </div>
      `,
    })

    return NextResponse.json({ success: true, data })
  } catch (error) {
    return NextResponse.json({ success: false, error }, { status: 500 })

import { useState, useEffect } from 'react'
import { supabase } from '@/lib/supabase'
import { useRouter } from 'next/navigation'
import Navbar from '@/components/Navbar'
import { calculateLevel } from '@/lib/levels'

type GachaState = 'idle' | 'scanning' | 'revealing' | 'result'

export default function GachaPage() {
  const router = useRouter()
  const [userProfile, setUserProfile] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  
  const [gachaState, setGachaState] = useState<GachaState>('idle')
  const [targetPlayer, setTargetPlayer] = useState<any>(null)
  const [rarity, setRarity] = useState<'N' | 'R' | 'SR' | 'SSR'>('N')
  
  // 每日次数限制
  const [remainingChances, setRemainingChances] = useState(5)
  const [maxChances, setMaxChances] = useState(5)

  // 关注状态
  const [isFollowing, setIsFollowing] = useState(false)
  const [followLoading, setFollowLoading] = useState(false)

  useEffect(() => {
    async function init() {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) { router.push('/login'); return }

      const { data: profile } = await supabase.from('users_profile').select('*').eq('auth_id', user.id).single()
      setUserProfile(profile)

      if (profile) {
        let limit = 5 
        if (profile.id === 1 || [10000, 11111, 22222, 33333, 44444, 55555, 66666, 77777, 88888, 99999].includes(profile.id)) {
          limit = 20 
        } else if (profile.level >= 50) {
          limit = 15
        } else if (profile.level >= 20) {
          limit = 10
        }
        setMaxChances(limit)

        const today = new Date().toISOString().split('T')[0]
        if (profile.last_gacha_at === today) {
          setRemainingChances(limit - (profile.daily_gacha_count || 0))
        } else {
          setRemainingChances(limit)
          await supabase.from('users_profile').update({ daily_gacha_count: 0, last_gacha_at: today }).eq('id', profile.id)
        }
      }
      setLoading(false)
    }
    init()
  }, [router])

  // 🔥 检查是否已经关注了抽到的人
  useEffect(() => {
    if (targetPlayer && userProfile) {
      supabase.from('follows')
        .select('id')
        .eq('follower_id', userProfile.id)
        .eq('following_id', targetPlayer.id)
        .single()
        .then(({ data }) => {
          setIsFollowing(!!data)
        })
    }
  }, [targetPlayer, userProfile])

  async function handleDraw() {
    if (remainingChances <= 0) return alert('今日盲盒次数已耗尽，升级靓号或明天再来吧！')
    
    setGachaState('scanning')
    setTargetPlayer(null)

    setTimeout(async () => {
      try {
        const { data: players, error } = await supabase
          .from('users_profile')
          .select('*')
          .neq('id', userProfile.id)
          .limit(20)

        if (error || !players || players.length === 0) {
          setGachaState('idle')
          return alert('雷达未扫描到其他玩家，请稍后再试！')
        }

        const randomPlayer = players[Math.floor(Math.random() * players.length)]
        setTargetPlayer(randomPlayer)

        const pLvl = calculateLevel(randomPlayer.experience || 0, randomPlayer.id).level
        let r: 'N' | 'R' | 'SR' | 'SSR' = 'N'
        if (pLvl > 60) r = 'SSR'
        else if (pLvl > 30) r = 'SR'
        else if (pLvl > 10) r = 'R'
        setRarity(r)

        const today = new Date().toISOString().split('T')[0]
        const newCount = (userProfile.daily_gacha_count || 0) + 1
        await supabase.from('users_profile').update({ daily_gacha_count: newCount, last_gacha_at: today }).eq('id', userProfile.id)
        
        setUserProfile({ ...userProfile, daily_gacha_count: newCount })
        setRemainingChances(prev => prev - 1)

        setGachaState('revealing')
        setTimeout(() => setGachaState('result'), 500)

      } catch (err) {
        setGachaState('idle')
      }
    }, 2000)
  }

  // 处理关注逻辑
  async function handleFollow() {
    if (!userProfile || !targetPlayer) return
    setFollowLoading(true)
    try {
      if (isFollowing) {
        await supabase.from('follows').delete().eq('follower_id', userProfile.id).eq('following_id', targetPlayer.id)
        setIsFollowing(false)
      } else {
        await supabase.from('follows').insert({ follower_id: userProfile.id, following_id: targetPlayer.id })
        await supabase.from('notifications').insert({ user_id: targetPlayer.id, actor_id: userProfile.id, type: 'follow' })
        setIsFollowing(true)
        alert('关注成功！可以去主页查看 TA 的常玩游戏了。')
      }
    } catch (err) {
      console.error(err)
    }
    setFollowLoading(false)
  }

  // 处理私信（关闭弹窗并唤起悬浮聊天窗）
  function handleSendMsg() {
    if (!targetPlayer) return
    setGachaState('idle')
    // 这里为了简化体验，直接跳转到带目标 ID 的消息页面。
    // 如果你想让右下角的悬浮窗直接弹出来并选中这个人，需要用到我们之前做好的 ChatWidget 事件，这里暂时先跳转。
    router.push(`/messages?target=${targetPlayer.id}`)
  }

  if (loading) return <div className="min-h-screen bg-[#050B14] flex items-center justify-center text-cyan-500 animate-pulse">雷达充能中...</div>

  return (
    <div className="min-h-screen bg-[#050B14] text-gray-200 relative overflow-hidden font-sans flex flex-col">
      {/* 雷达背景特效 */}
      <div className="fixed inset-0 z-0 pointer-events-none">
        <div className="absolute inset-0 bg-[url('/bg.png')] bg-cover bg-center bg-no-repeat opacity-40"></div>
        <div className="absolute inset-0 bg-gradient-to-b from-[#050B14]/80 via-[#050B14]/60 to-[#050B14]"></div>
      </div>

      <Navbar />

  const searchParams = useSearchParams()

  const gameId = searchParams.get('id')

  const [game, setGame] = useState<any>(null)
  const [subGames, setSubGames] = useState<any[]>([])
  const [userProfile, setUserProfile] = useState<any>(null)
  const [recruits, setRecruits] = useState<any[]>([])
  const [posts, setPosts] = useState<any[]>([])
  const [followedRecruitIds, setFollowedRecruitIds] = useState<number[]>([])
  const [likedIds, setLikedIds] = useState<number[]>([])
  const [isFollowingGame, setIsFollowingGame] = useState(false)
  const [loading, setLoading] = useState(true)

  const [activeTab, setActiveTab] = useState<'recruit' | 'forum'>('recruit')

  const [isPostingRecruit, setIsPostingRecruit] = useState(false)
  const [isPostingForum, setIsPostingForum] = useState(false)
  const [activeRecruit, setActiveRecruit] = useState<any>(null)
  const [activePost, setActivePost] = useState<any>(null)
  const [comments, setComments] = useState<any[]>([])
  const [postComments, setPostComments] = useState<any[]>([])
  const [commentLoading, setCommentLoading] = useState(false)

  useEffect(() => {
    async function init() {
      if (!gameId) return
      try {
        const { data: { user } } = await supabase.auth.getUser()
        if (user) {
          const { data } = await supabase.from('users_profile').select('*').eq('auth_id', user.id).single()
          if (data) {
            setUserProfile(data)
            const { data: l } = await supabase.from('likes').select('recruit_id').eq('user_id', data.id)
            if (l) setLikedIds(l.map(x => x.recruit_id))
            const { data: rf } = await supabase.from('recruit_follows').select('recruit_id').eq('user_id', data.id)
            if (rf) setFollowedRecruitIds(rf.map(x => x.recruit_id))
            const { data: gf } = await supabase.from('game_follows').select('id').eq('user_id', data.id).eq('game_id', gameId).single()
            if (gf) setIsFollowingGame(true)
          }
        }

        const { data: gameData } = await supabase.from('games').select(`*, creator:users_profile(nickname)`).eq('id', gameId).single()
        setGame(gameData)

        if (gameData && !gameData.parent_id) {
          const { data: subs } = await supabase.from('games').select(`*`).eq('parent_id', gameId).order('created_at', { ascending: false })
          if (subs) setSubGames(subs)
        }
        await fetchRecruits()
        await fetchPosts()
      } catch (err) {} finally { setLoading(false) }
    }
    init()
  }, [gameId])

  async function fetchRecruits() {
    const { data } = await supabase.from('recruits').select(`*, creator:users_profile(nickname, avatar_url, id), likes(id)`).eq('game_id', gameId).order('created_at', { ascending: false })
    if (data) setRecruits(data)
  }

  async function fetchPosts() {
    const { data } = await supabase.from('posts').select(`*, user:users_profile(nickname, avatar_url, id)`).eq('game_id', gameId).order('created_at', { ascending: false })
    if (data) setPosts(data)
  }

  async function toggleFollowGame() {
    if (!userProfile) return alert('请先登录！')
    if (isFollowingGame) {
      await supabase.from('game_follows').delete().eq('user_id', userProfile.id).eq('game_id', gameId)
      setIsFollowingGame(false)
    } else {
      await supabase.from('game_follows').insert({ user_id: userProfile.id, game_id: gameId })
      setIsFollowingGame(true)
    }
  }

  async function toggleFollowRecruit(recruitId: number) {
    if (!userProfile) return alert('请先登录！')
    if (followedRecruitIds.includes(recruitId)) {
      await supabase.from('recruit_follows').delete().eq('user_id', userProfile.id).eq('recruit_id', recruitId)
      setFollowedRecruitIds(followedRecruitIds.filter(id => id !== recruitId))
    } else {
      await supabase.from('recruit_follows').insert({ user_id: userProfile.id, recruit_id: recruitId })
      setFollowedRecruitIds([...followedRecruitIds, recruitId])
    }
  }

  // 点赞：走 RPC，防重复 / 每日限次 / 双方加经验全在数据库里完成
  async function handleLike(recruitId: number, creatorId: number) {
    if (!userProfile) return alert('请先登录！')
    if (likedIds.includes(recruitId)) return alert('你已经赞过这条通缉令啦！')
    if (creatorId === userProfile.id) return alert('不能给自己点赞哦！')

    const { error } = await supabase.rpc('give_like', { p_recruit_id: recruitId })

    if (error) {
      alert(error.message)
      return
    }

    const today = new Date().toISOString().split('T')[0]
    const currentLikeCount = userProfile.last_like_at === today ? (userProfile.daily_like_count || 0) : 0

    setLikedIds([...likedIds, recruitId])
    setUserProfile({
      ...userProfile,
      experience: (userProfile.experience || 0) + 1,
      last_like_at: today,
      daily_like_count: currentLikeCount + 1,
    })
    fetchRecruits()
  }

  async function onRecruitSubmit(data: any) {
    if (!userProfile) {
      alert('错误：未获取到当前用户信息')
      return
    }

    const { error } = await supabase.from('recruits').insert({
      game_id: gameId,
      creator_id: userProfile.id,
      play_content: data.playContent,
      online_time: data.onlineTime,
      personality: data.personality,
      platform: data.platform,
      server_region: data.serverRegion
    })

    if (error) {
      alert('发布失败：' + error.message)
    } else {
      setIsPostingRecruit(false)
      fetchRecruits()
    }
  }

  async function onForumSubmit(data: any) {
    const { error } = await supabase.from('posts').insert({ game_id: gameId, user_id: userProfile.id, title: data.title, content: data.content })
    if (!error) { setIsPostingForum(false); fetchPosts() }
  }

    setActiveRecruit(recruit)
    setCommentLoading(true)
    const { data } = await supabase.from('comments').select(`*, user:users_profile(nickname, avatar_url)`).eq('recruit_id', recruit.id).order('created_at', { ascending: true })
    if (data) setComments(data)
    setCommentLoading(false)
  }

  async function sendComment(content: string) {
    await supabase.from('comments').insert({ recruit_id: activeRecruit.id, user_id: userProfile.id, content })
    openComments(activeRecruit)
  }

    setCommentLoading(false)

  return (

    if (!account || !password) return setMsg('请输入账号和密码')
    setLoading(true); setMsg('')
    
    try {
      let loginEmail = account
      if (!account.includes('@')) {
        const isId = /^\d+$/.test(account)
        const { data, error } = await supabase.from('users_profile').select('email').or(isId ? `id.eq.${account}` : `username.eq.${account}`).single()
        if (error || !data) {
          setMsg('找不到该用户'); setLoading(false); return 
        }
        loginEmail = data.email
      }
      
      const { data: authData, error: loginError } = await supabase.auth.signInWithPassword({ email: loginEmail, password })
      
      if (loginError) {
        setLoading(false)
        setMsg('登录失败：账号或密码错误')
        return
      }

      // 🔥 检查封禁状态
      if (authData.user) {
        const { data: profile } = await supabase.from('users_profile').select('banned_until').eq('auth_id', authData.user.id).single()
        
        if (profile && profile.banned_until && new Date(profile.banned_until) > new Date()) {
          const banDate = new Date(profile.banned_until)
          const diffDays = Math.ceil((banDate.getTime() - new Date().getTime()) / (1000 * 3600 * 24))
          
          let banText = diffDays > 3650 ? '永久' : `${diffDays} 天`
          
          await supabase.auth.signOut()
          setLoading(false)
          setMsg(`您已经被巡游天官发现您的违禁行为，禁止进入 (剩余封禁: ${banText})`)
          return
        }
      }

      setLoading(false)
      setMsg('登录成功！正在跳转...')
      window.location.href = '/'
      
    } catch (err: any) {
      setLoading(false)
      setMsg('请求超时，请检查网络')
    }
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-[#050B14] p-4">

function timeAgo(iso: string) {
  const diff = Date.now() - new Date(iso).getTime()
  const m = Math.floor(diff / 60000)
  if (m < 1) return '刚刚'
  if (m < 60) return `${m} 分钟前`
  const h = Math.floor(m / 60)
  if (h < 24) return `${h} 小时前`
  const d = Math.floor(h / 24)
  if (d < 30) return `${d} 天前`
  return new Date(iso).toLocaleDateString('zh-CN')
}

function timeAgo(iso: string) {
  const diff = Date.now() - new Date(iso).getTime()
  const m = Math.floor(diff / 60000)
  if (m < 1) return '刚刚'
  if (m < 60) return `${m} 分钟前`
  const h = Math.floor(m / 60)
  if (h < 24) return `${h} 小时前`
  const d = Math.floor(h / 24)
  if (d < 30) return `${d} 天前`
  return new Date(iso).toLocaleDateString('zh-CN')
}

    }

  }, [myTeam, activeTab])

  async function fetchTeams(keyword = '') {
    let query = supabase.from('teams').select('*').order('created_at', { ascending: false }).limit(20)
    if (keyword) query = query.ilike('name', `%${keyword}%`)

  }

  async function fetchMembers(teamId: number) {
    const { data } = await supabase.from('team_members').select(`role, users_profile(id, nickname, avatar_url)`).eq('team_id', teamId).neq('role', 'pending')
    if (data) {
      const roleWeight: any = { leader: 4, vice_leader: 3, admin: 2, member: 1 }
      setMembers(data.sort((a, b) => roleWeight[b.role] - roleWeight[a.role]))
    }
  }

  async function fetchPending(teamId: number) {
    const { data } = await supabase.from('team_members').select(`id, users_profile(id, nickname, avatar_url)`).eq('team_id', teamId).eq('role', 'pending')
    if (data) setPendingMembers(data)
  }

  async function fetchMessages(teamId: number, isSilent = false) {
    const { data } = await supabase.from('team_messages').select(`*, user:users_profile(nickname, avatar_url)`).eq('team_id', teamId).order('created_at', { ascending: true })
    if (data) {
      setMessages(data)
      if (isSilent && data.length > prevMsgLength.current) {
        setTimeout(() => bottomRef.current?.scrollIntoView({ behavior: 'smooth' }), 100)
      }
      prevMsgLength.current = data.length
    }
  }

    if (!newTeamName || !userProfile) return setCreateMsg('请填写名称')
    setCreateMsg('创建中...')

    if (newTeamIcon) {

        iconUrl = data.publicUrl
      }

    if (teamErr) return setCreateMsg('创建失败，名字可能已存在')

    setMyTeam(newTeam); setMyRole('leader'); setIsCreating(false)
  }

  async function handleApply(teamId: number) {
    if (!userProfile) return
    const { error } = await supabase.from('team_members').insert({ team_id: teamId, user_id: userProfile.id, role: 'pending' })

  const [msg, setMsg] = useState('')
  const [loading, setLoading] = useState(false)
  
  const [avatarFile, setAvatarFile] = useState<File | null>(null)
  const [avatarPreview, setAvatarPreview] = useState('')

  // 🔥 成就与称号状态
  const [myAchievements, setMyAchievements] = useState<any[]>([])

  useEffect(() => {
    supabase.auth.getUser().then(({ data }) => {
      if (data.user) {
        supabase.from('users_profile').select('*').eq('auth_id', data.user.id).single().then(({ data: p }) => {
          if (p) {
            setProfile(p)
            setNickname(p.nickname || '')
            setAvatarPreview(p.avatar_url || '')
            fetchAchievements(p.id)
          }
        })
      } else {
        router.push('/login')
      }
    })
  }, [router])

  async function fetchAchievements(userId: number) {
    // 联表查询：查出该用户解锁的所有成就
    const { data } = await supabase
      .from('user_achievements')
      .select(`*, achievement:achievements(*)`)
      .eq('user_id', userId)
    
    if (data) setMyAchievements(data)
  }

  const handleAvatarChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files && e.target.files[0]) {
      const file = e.target.files[0]
      if (file.size > 500 * 1024) {
        alert('头像图片不能超过 500KB，请压缩后再上传！')
        e.target.value = ''
        return
      }
      setAvatarFile(file)
      setAvatarPreview(URL.createObjectURL(file))
    }
  }

  async function updateProfile(e: any) {
    e.preventDefault()
    setLoading(true)
    setMsg('')

    const newNickname = e.target.nickname.value.trim()
    if (newNickname.length === 0 || newNickname.length > 9) {
      setMsg('昵称长度必须在 1-9 个字符之间'); setLoading(false); return
    }

    if (newNickname !== profile.nickname) {
      const { data: exist } = await supabase.from('users_profile').select('id').eq('nickname', newNickname).neq('id', profile.id).single()
      if (exist) {
        setMsg('该昵称已被其他玩家使用'); setLoading(false); return
      }
    }

    let newAvatarUrl = profile.avatar_url
    if (avatarFile) {
      const fileName = `avatar-${profile.id}-${Date.now()}.${avatarFile.name.split('.').pop()}`
      const { error } = await supabase.storage.from('avatars').upload(fileName, avatarFile)
      if (!error) {
        const { data } = supabase.storage.from('avatars').getPublicUrl(fileName)
        newAvatarUrl = data.publicUrl
      }
    }

    const { error } = await supabase.from('users_profile').update({
      nickname: newNickname,
      qq: e.target.qq.value,
      wechat: e.target.wechat.value,
      yy: e.target.yy.value,
      avatar_url: newAvatarUrl
    }).eq('id', profile.id)

    setLoading(false)
    if (error) {
      setMsg('更新失败：' + error.message)
    } else {
      setMsg('资料更新成功！')
      setProfile({ ...profile, nickname: newNickname, qq: e.target.qq.value, wechat: e.target.wechat.value, yy: e.target.yy.value, avatar_url: newAvatarUrl })
      router.refresh()
    }
  }

  async function updatePassword() {
    if (!newPassword) return
    if (newPassword.length < 6) return setMsg('密码长度至少6位')
    setLoading(true)
    const { error } = await supabase.auth.updateUser({ password: newPassword })
    setLoading(false)
    if (error) setMsg(error.message)
    else {
      setMsg('密码修改成功！')
      setNewPassword('')
    }
  }

  // 🔥 佩戴称号
  async function equipTitle(title: string | null) {
    setLoading(true)

      alert(title ? `成功佩戴称号：${title}` : '已隐藏当前称号')
    }
  }

  if (!profile) return <div className="min-h-screen bg-[#050B14]"></div>

  const levelInfo = calculateLevel(profile.experience || 0, profile.id)
  const progressPercent = levelInfo.isMaxLevel ? 100 : Math.min(100, Math.floor((levelInfo.currentLevelExp / levelInfo.nextLevelExp) * 100))

  return (
    <div className="min-h-screen bg-[#050B14] text-gray-200 pb-20">
      <Navbar />
      <div className="max-w-2xl mx-auto p-6 mt-10 bg-[#0A111F]/80 border border-cyan-500/30 rounded-2xl shadow-[0_0_30px_rgba(6,182,212,0.15)] backdrop-blur-md">
        <h1 className="text-2xl font-black text-cyan-400 mb-6 border-b border-gray-800 pb-4 font-orbitron">用户中心</h1>

import { useState, useEffect, useRef } from 'react'
import { supabase } from '@/lib/supabase'

export default function ChatWidget() {
  const [isOpen, setIsOpen] = useState(false)
  const [userProfile, setUserProfile] = useState<any>(null)
  const [contacts, setContacts] = useState<any[]>([])
  const [following, setFollowing] = useState<any[]>([]) // 我关注的所有人
  const [activeContact, setActiveContact] = useState<any>(null)
  const [messages, setMessages] = useState<any[]>([])
  const [newMsg, setNewMsg] = useState('')
  
  // 分页状态：chat(最近聊天) 或 following(我的关注)
  const [viewMode, setViewMode] = useState<'chat' | 'following'>('chat')
  
  const bottomRef = useRef<HTMLDivElement>(null)
  const prevMsgLength = useRef(0)

  useEffect(() => {
    const handleOpen = () => setIsOpen(true)
    window.addEventListener('openChatWidget', handleOpen)

    async function init() {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) return
      const { data: profile } = await supabase.from('users_profile').select('*').eq('auth_id', user.id).single()
      setUserProfile(profile)
      
      if (profile) {
        // 1. 获取互关好友（用于最近聊天列表）
        const { data: followingData } = await supabase.from('follows').select('following_id').eq('follower_id', profile.id)
        const { data: followersData } = await supabase.from('follows').select('follower_id').eq('following_id', profile.id)
        
        let mutualIds: number[] = []
        if (followingData && followersData) {
          const followingIds = followingData.map(f => f.following_id)
          const followerIds = followersData.map(f => f.follower_id)
          mutualIds = followingIds.filter(id => followerIds.includes(id))
        }

        if (mutualIds.length > 0) {
          const { data: contactsData } = await supabase.from('users_profile').select('*').in('id', mutualIds)
          if (contactsData) setContacts(contactsData)
        }

        // 2. 获取我关注的所有人（用于"我的关注"列表）
        if (followingData && followingData.length > 0) {
          const allFollowingIds = followingData.map(f => f.following_id)
          const { data: allFollowingData } = await supabase.from('users_profile').select('*').in('id', allFollowingIds)
          if (allFollowingData) setFollowing(allFollowingData)
        }
      }
    }
    init()

    return () => window.removeEventListener('openChatWidget', handleOpen)
  }, [])

  useEffect(() => {
    let interval: NodeJS.Timeout
    if (activeContact && userProfile && isOpen) {
      fetchMessages(activeContact.id, false)
      interval = setInterval(() => fetchMessages(activeContact.id, true), 3000)
    }
    return () => { if (interval) clearInterval(interval) }
  }, [activeContact, userProfile, isOpen])

  async function fetchMessages(contactId: number, isSilent = false) {
    if (!userProfile) return
    const { data } = await supabase.from('messages').select('*')
      .or(`and(sender_id.eq.${userProfile.id},receiver_id.eq.${contactId}),and(sender_id.eq.${contactId},receiver_id.eq.${userProfile.id})`)
      .order('created_at', { ascending: true })
    
    if (data) {
      if (data.length !== prevMsgLength.current || !isSilent) {
        setMessages(data)
        if (isSilent && data.length > prevMsgLength.current) {
          setTimeout(() => bottomRef.current?.scrollIntoView({ behavior: 'smooth' }), 100)
        }
        prevMsgLength.current = data.length
        await supabase.from('messages').update({ is_read: true }).eq('sender_id', contactId).eq('receiver_id', userProfile.id).eq('is_read', false)
      }
    }
  }

  async function handleSend() {
    if (!newMsg.trim() || !activeContact || !userProfile) return
    const content = newMsg
    setNewMsg('')
    setMessages(prev => [...prev, { id: Date.now(), sender_id: userProfile.id, receiver_id: activeContact.id, content, created_at: new Date().toISOString(), is_read: false }])
    setTimeout(() => bottomRef.current?.scrollIntoView({ behavior: 'smooth' }), 50)
    
    await supabase.from('messages').insert({ sender_id: userProfile.id, receiver_id: activeContact.id, content })
  }

  function selectContact(contact: any) {
    setActiveContact(contact)
    prevMsgLength.current = 0
  }

  if (!isOpen) return null

  // 根据当前 Tab 决定展示的列表
  const displayList = viewMode === 'chat' ? contacts : following

  return (
    <div className="fixed bottom-6 right-6 w-80 h-[500px] bg-[#0A111F] border border-cyan-500/50 rounded-2xl shadow-[0_0_30px_rgba(6,182,212,0.3)] flex flex-col z-[999] overflow-hidden animate-in slide-in-from-bottom-5">
      {/* 头部 */}
      <div className="bg-[#050B14] p-3 border-b border-gray-800 flex justify-between items-center">
        <h3 className="text-cyan-400 font-bold font-orbitron text-sm">✉️ 战术通讯</h3>
        <button onClick={() => setIsOpen(false)} className="text-gray-500 hover:text-red-500 text-xl leading-none">&times;</button>
      </div>

      <div className="flex-1 flex overflow-hidden">
        {/* 左侧列表区域 */}
        <div className={`w-1/3 border-r border-gray-800 bg-[#050B14]/50 flex flex-col ${activeContact ? 'hidden md:flex' : 'flex'}`}>
          {/* 分页 Tab */}
          <div className="flex border-b border-gray-800 bg-black/40">
            <button 
              onClick={() => { setViewMode('chat'); setActiveContact(null); }} 
              className={`flex-1 py-2 text-[10px] font-bold ${viewMode === 'chat' ? 'text-cyan-400 border-b-2 border-cyan-500' : 'text-gray-500'}`}
            >
              聊天
            </button>
            <button 
              onClick={() => { setViewMode('following'); setActiveContact(null); }} 
              className={`flex-1 py-2 text-[10px] font-bold ${viewMode === 'following' ? 'text-cyan-400 border-b-2 border-cyan-500' : 'text-gray-500'}`}
            >
              关注
            </button>
          </div>

          <div className="flex-1 overflow-y-auto">
            {displayList.length === 0 ? (
               <p className="text-center text-gray-600 text-[10px] mt-4 px-2">
                 {viewMode === 'chat' ? '暂未建立互关聊天' : '还没有关注任何人'}
               </p>
            ) : (
              displayList.map(c => (
                <div key={c.id} onClick={() => selectContact(c)} className="p-2 flex flex-col items-center cursor-pointer hover:bg-cyan-900/20 border-b border-gray-800/50">
                  <img src={c.avatar_url || '/logo.png'} className="w-10 h-10 rounded-full border border-gray-700 object-cover bg-black" />
                  <span className="text-[10px] text-gray-300 mt-1 truncate w-full text-center">{c.nickname}</span>

          {activeContact ? (
            <>
              <div className="p-2 bg-black/40 border-b border-gray-800 flex items-center gap-2">

'use client'

import { useState } from 'react'
import { supabase } from '@/lib/supabase'

export default function FeedbackButton() {
  const [isOpen, setIsOpen] = useState(false)
  const [type, setType] = useState('建议')
  const [content, setContent] = useState('')
  const [contact, setContact] = useState('')
  const [loading, setLoading] = useState(false)
  const [msg, setMsg] = useState('')

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    if (!content.trim()) return setMsg('请填写反馈内容')
    setLoading(true); setMsg('')

    try {
      // 1. 将反馈内容存入数据库
      const { data: { user } } = await supabase.auth.getUser()
      let userId = null
      if (user) {
        const { data: profile } = await supabase.from('users_profile').select('id').eq('auth_id', user.id).single()
        if (profile) userId = profile.id
      }

      await supabase.from('feedbacks').insert({
        user_id: userId, 
        type, 
        content, 
        contact
      })

      // 2. 调用后端 API 发送邮件
      const res = await fetch('/api/feedback', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ type, content, contact })
      })

      if (res.ok) {
        setMsg('提交成功！感谢你的反馈！')
        setContent(''); setContact('')
        setTimeout(() => setIsOpen(false), 2000)
      } else {
        setMsg('已存入数据库，但邮件发送失败。')
      }
    } catch (err) {
      setMsg('网络异常，请稍后再试')
    } finally {
      setLoading(false)
    }
  }

  return (
    <>
      {/* 悬浮按钮 */}
      <button 
        onClick={() => setIsOpen(true)}

import { useState, useEffect } from 'react'
import { supabase } from '@/lib/supabase'

export default function FollowUserButton({ targetId, currentUserId }: { targetId: number, currentUserId?: number }) {
  const [isFollowing, setIsFollowing] = useState(false)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!currentUserId || currentUserId === targetId) { setLoading(false); return }
    supabase.from('follows').select('id').eq('follower_id', currentUserId).eq('following_id', targetId).single().then(({ data }) => {
      setIsFollowing(!!data)
      setLoading(false)
    })
  }, [currentUserId, targetId])

  async function toggleFollow(e: React.MouseEvent) {
    e.preventDefault()
    e.stopPropagation()
    if (!currentUserId) return alert('请先登录！')
    if (currentUserId === targetId) return

    if (isFollowing) {
      await supabase.from('follows').delete().eq('follower_id', currentUserId).eq('following_id', targetId)
      setIsFollowing(false)
    } else {
      await supabase.from('follows').insert({ follower_id: currentUserId, following_id: targetId })
      await supabase.from('notifications').insert({ user_id: targetId, actor_id: currentUserId, type: 'follow' })
      setIsFollowing(true)
    }
  }

  if (loading || !currentUserId || currentUserId === targetId) return null

  return (
    <button 
      onClick={toggleFollow} 
      className={`px-3 py-1 rounded text-xs font-bold transition-all border ${
        isFollowing 
          ? 'bg-cyan-900/40 text-cyan-400 border-cyan-500/50' 
          : 'bg-transparent text-blue-400 border-blue-500/50 hover:bg-blue-500/20'
      }`}
    >
      {isFollowing ? '已关注' : '+ 关注'}
    </button>
  )
}

import { useState, useEffect, useRef } from 'react'
import { supabase } from '@/lib/supabase'
import Link from 'next/link'
import { useRouter, usePathname } from 'next/navigation'
import { calculateLevel } from '@/lib/levels'

export default function Navbar() {
  const router = useRouter()
  const pathname = usePathname()
  const [userProfile, setUserProfile] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  const [isMenuOpen, setIsMenuOpen] = useState(false)
  
  const [unreadNotif, setUnreadNotif] = useState(0)
  const [hasUnreadMsg, setHasUnreadMsg] = useState(false)

  const menuRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    fetchUser()
    
    function handleClickOutside(event: MouseEvent) {
      if (menuRef.current && !menuRef.current.contains(event.target as Node)) {
        setIsMenuOpen(false)
      }
    }
    document.addEventListener("mousedown", handleClickOutside)
    return () => document.removeEventListener("mousedown", handleClickOutside)
  }, [pathname])

  useEffect(() => {
    let interval: NodeJS.Timeout
    if (userProfile) {
      fetchUnread()
      interval = setInterval(fetchUnread, 5000)
    }
    return () => { if (interval) clearInterval(interval) }
  }, [userProfile, pathname])

  async function fetchUser() {
    const { data: { user } } = await supabase.auth.getUser()
    if (user) {
      const { data } = await supabase.from('users_profile').select('*').eq('auth_id', user.id).single()
      if (data) {
        if (data.banned_until && new Date(data.banned_until) > new Date()) {
          alert('您已经被巡游天官发现您的违禁行为，禁止进入！')
          await supabase.auth.signOut()
          window.location.href = '/login'
          return
        }
        setUserProfile(data)
      }
    }
    setLoading(false)
  }

  async function fetchUnread() {
    if (!userProfile) return
    const { count: notifCount } = await supabase.from('notifications').select('*', { count: 'exact', head: true }).eq('user_id', userProfile.id).eq('is_read', false)
    setUnreadNotif(notifCount || 0)

    const { count: msgCount } = await supabase.from('messages').select('*', { count: 'exact', head: true }).eq('receiver_id', userProfile.id).eq('is_read', false)
    setHasUnreadMsg((msgCount || 0) > 0)
  }

  async function handleLogout() {
    await supabase.auth.signOut()
    window.location.href = '/'

import { createClient } from '@supabase/supabase-js'

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!
const supabaseKey = process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!

// 核心：使用 Promise.race 强制 5 秒超时，防止 Cloudflare Worker 卡死
const fetchWithTimeout = async (url: RequestInfo | URL, options: RequestInit = {}) => {
  const timeout = new Promise<Response>((_, reject) => {
    setTimeout(() => reject(new Error("⏰ Supabase 请求超时 (5秒)")), 5000)
  })
  
  return Promise.race([
    fetch(url, options),
    timeout
  ]) as Promise<Response>
}

export const supabase = createClient(supabaseUrl, supabaseKey, {
  global: {
    fetch: fetchWithTimeout,
  },
})
